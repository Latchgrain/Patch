+# Latchgrain pilot changes
+
+This patch turns the SQLite import into a working query engine and ships the
+2024 Toyota VIN Reference as the first real dataset, at $0.00 per query.
+
+## What changed
+
+**Imported datasets can now be queried.**
+- `ProviderDatasetImport.sourceDatabase` (new, nullable) keeps the uploaded
+  SQLite file on the server. `ProviderDatasetImportTable.queryableColumns` (new)
+  lists which columns buyers may filter on.
+- New endpoint `GET/POST /api/datasets/:slug/query`:
+  - `GET` returns the queryable tables and columns, the price, and the caller's
+    access state and remaining queries.
+  - `POST {table, filters, responseMode}` runs a parameterized, exact-match
+    (case-insensitive) filter query.
+  - Response modes: `answer_only` (exactly one match, otherwise "ambiguous"),
+    `summarized_data` (counts of distinct values per column), and `raw_records`
+    (up to 25 rows, flagged when truncated).
+  - Every result carries provenance. If the table has a `source_id` column and
+    the database has a `sources` table, those sources are cited with their URLs.
+  - Callers sign in or use an approved AI-application API key. The dataset
+    owner and admins can always query. Everyone else needs an APPROVED access
+    request.
+  - Each query is written to the usage ledger with its price and revenue share.
+    The ledger stores counts and provenance only, never the returned rows.
+- The dataset detail page shows a query form (table, filters, mode, results,
+  sources) to anyone allowed to query, and an access notice to everyone else.
+- The admin SQLite import now stores the file, makes every column filterable by
+  default, and sets the importing admin as owner. Before this, imported
+  datasets had no owner, so the ledger silently skipped every query.
+
+**Bugs fixed**
+- The query cap is now per requester. It used to count every buyer's queries
+  together, so 100 total queries shut the dataset off for everyone. The
+  anonymous homepage demo keeps a dataset-wide cap.
+- The dataset owner and admins no longer need to approve their own access
+  request before querying.
+- Only admins can change revenue share. Providers previously could set their own
+  share, up to 100%, and overwrite the admin's value.
+- Query history no longer fails completely when one stored row is unreadable.
+  That row is skipped.
+- The provider workspace hides the automotive-only filter toggles on other
+  datasets.
+- `sql.js` is marked as a server-external package, so its WebAssembly file
+  loads reliably in production.
+
+**First dataset**
+- `data/toyota_2024_vin_reference.sqlite` is seeded on every boot, idempotently,
+  as "2024 Toyota VIN Reference" by "Latchgrain Open Data":
+  - Published and approved, at $0.00 per query, with 1,000 queries per requester.
+  - Owned by the `POLSIA_OWNER_EMAIL` account, or else the oldest verified admin.
+  - Queryable tables: `models`, `powertrain_configurations`,
+    `model_configuration_summary`, `vin_position_rules`, `vpic_decoder_fields`,
+    `wmi_codes`, and `sources`.
+  - Later admin edits to price, name, or status are not overwritten.
+
+## Database
+
+The changes are additive only: one nullable column and one column with a
+default. Polsia's startup `prisma db push` applies them. No data migration is
+needed.
+
+## Verification
+
+- Typecheck is clean and the production build succeeds.
+- 589/589 unit tests pass, including 20 new tests:
+  - Engine tests run real queries against the bundled Toyota database, with no
+    mocks.
+  - Route tests cover access rules, the per-requester cap, ledger shape, and
+    API keys.
+- Not run: the Postgres integration tests and the Playwright browser tests.
+  These need a live database. Check them after deploying.
+
+## After deploying: smoke test
+
+1. Sign in as the owner account and open `/datasets/toyota-2024-vin-reference`.
+2. Query table `models` with `model_name = Prius` in Answer only mode. You
+   should get one record with an NHTSA source link.
+3. Query table `vin_position_rules` with `position_start = 10`. You should get
+   `code_for_2024 = R`.
+4. Check `/admin/usage-ledger`. Both queries should appear at $0.00.
+5. From a second account, request access, approve it, run a query, and check
+   that the remaining-queries count goes down.

+//
+// GET  /api/datasets/:slug/query  → queryable tables/columns + caller's access state
+// POST /api/datasets/:slug/query  → run a filter query
+//
+// Authentication: a signed-in session, or `Authorization: Bearer <api key>` for an
+// approved AI application. The dataset owner and administrators can always
+// query; everyone else needs an APPROVED access request for this dataset.

+function errorResponse(status: number, code: string, message: string) {
+  return NextResponse.json({ error: { code, message } }, { status });

+const notFound = () =>
+  errorResponse(404, 'NOT_FOUND', 'This dataset is not available for queries.');
+const internalError = () => errorResponse(500, 'INTERNAL_ERROR', 'Unable to process the query.');

+async function loadDataset(slug: string) {
+  const dataset = await prisma.marketplaceDataset.findUnique({
+    where: { detailSlug: slug },
+    select: {
+      id: true,
+      stableKey: true,
+      name: true,
+      provider: true,
+      accessType: true,
+      queryPriceCents: true,
+      revenueSharePercent: true,
+      maxQueries: true,
+      publicationStatus: true,
+      approvalStatus: true,
+      ownerId: true,
+    },
+  });
+  if (!dataset || dataset.accessType !== 'Query API') return null;
+  const imported = await prisma.providerDatasetImport.findUnique({
+    where: { datasetStableKey: dataset.stableKey },
+    include: { tables: { orderBy: { name: 'asc' } } },
+  });
+  if (!imported?.sourceDatabase) return null;
+  return { dataset, imported };

+type Loaded = NonNullable<Awaited<ReturnType<typeof loadDataset>>>;

+type Principal = { userId: string; role: string | null; viaApiKey: boolean };

+// Returns null for anonymous callers. Throws a Response for an invalid API key.
+async function resolvePrincipal(request: Request): Promise<Principal | null> {
+  const apiKey = await authorizeApiKey(request);
+  if (apiKey) {
+    const user = await prisma.user.findUnique({
+      where: { id: apiKey.userId },
+      select: { role: true },
+    });
+    return { userId: apiKey.userId, role: user?.role ?? null, viaApiKey: true };
+  }
+  const user = await getSessionUser();
+  if (!user) return null;
+  return {
+    userId: user.id,
+    role: (user as { role?: string | null }).role ?? null,
+    viaApiKey: false,
+  };
+}
+
+async function resolveAccess(
+  loaded: Loaded,
+  principal: Principal | null,
+): Promise<ImportedQueryAccess> {
+  if (!principal) return 'sign_in_required';
+  if (loaded.dataset.ownerId === principal.userId) return 'owner';
+  if (principal.role === 'admin') return 'admin';
+  if (
+    !principal.viaApiKey &&
+    !(await hasAccountCapability(principal.userId, AccountCapabilityType.END_USER))
+  ) {
+    return 'capability_required';
+  }
+  const accessRequest = await prisma.datasetAccessRequest.findUnique({
+    where: { datasetId_userId: { datasetId: loaded.dataset.id, userId: principal.userId } },
+    select: { status: true },
+  });
+  return accessRequest?.status === 'APPROVED' ? 'approved' : 'request_required';
+}
+
+function isPublic(loaded: Loaded) {
+  return (
+    loaded.dataset.publicationStatus === 'PUBLISHED' && loaded.dataset.approvalStatus === 'APPROVED'
+  );
+}
+
+// Per-requester cap: each requester gets `maxQueries` successful queries.
+// Owners and administrators are not capped.
+async function remainingQueries(loaded: Loaded, principal: Principal, access: ImportedQueryAccess) {
+  if (access !== 'approved') return null;
+  const used = await prisma.usageLedger.count({
+    where: { datasetId: loaded.dataset.id, userId: principal.userId, outcome: 'success' },
+  });
+  return Math.max(loaded.dataset.maxQueries - used, 0);
+}

+function catalogFor(loaded: Loaded) {
+  return loaded.imported.tables.map((table) => ({
+    name: table.name,
+    rowCount: table.rowCount,
+    columns: Array.isArray(table.columns)
+      ? table.columns.flatMap((column) =>
+          column &&
+          typeof column === 'object' &&
+          'name' in column &&
+          typeof column.name === 'string'
+            ? [column.name]
+            : [],
+        )
+      : [],
+    queryableColumns: table.queryableColumns,
+  }));
+}

+export async function GET(request: Request, { params }: RouteContext) {
+  const { slug } = await params;
+  try {
+    const loaded = await loadDataset(slug);
+    if (!loaded) return notFound();
+
+    let principal: Principal | null;
+    try {
+      principal = await resolvePrincipal(request);
+    } catch (response) {
+      if (response instanceof Response) return response;
+      throw response;
+    }
+    const access = await resolveAccess(loaded, principal);
+    if (!isPublic(loaded) && access !== 'owner' && access !== 'admin') return notFound();
+
+    return NextResponse.json(
+      ImportedDatasetSchema.parse({
+        dataset: {
+          stableKey: loaded.dataset.stableKey,
+          name: loaded.dataset.name,
+          provider: loaded.dataset.provider,
+        },
+        source: {
+          filename: loaded.imported.sourceFilename,
+          importedAt: loaded.imported.createdAt.toISOString(),
+        },
+        tables: catalogFor(loaded).filter((table) => table.queryableColumns.length > 0),
+        responseModes: ImportedResponseMode.options,
+        rawRecordLimit: IMPORTED_RAW_RECORD_LIMIT,
+        queryPriceCents: loaded.dataset.queryPriceCents,
+        access,
+        remainingQueries: principal ? await remainingQueries(loaded, principal, access) : null,
+      }),
+    );
+  } catch {
+    return internalError();
+  }

+export async function POST(request: Request, { params }: RouteContext) {
+  const { slug } = await params;
+  const parsed = ImportedQueryRequest.safeParse(await request.json().catch(() => null));
+
+  try {
+    const loaded = await loadDataset(slug);
+    if (!loaded) return notFound();
+
+    let principal: Principal | null;
+    try {
+      principal = await resolvePrincipal(request);
+    } catch (response) {
+      if (response instanceof Response) return response;
+      throw response;
+    }
+    const access = await resolveAccess(loaded, principal);
+    if (!isPublic(loaded) && access !== 'owner' && access !== 'admin') return notFound();
+    if (!principal || access === 'sign_in_required') {
+      return errorResponse(401, 'UNAUTHORIZED', 'Sign in or use an API key to query this dataset.');
+    }
+    if (access === 'capability_required') {
+      return errorResponse(403, 'FORBIDDEN', 'This account capability is required.');
+    }
+    if (access === 'request_required') {
+      return errorResponse(
+        403,
+        'DATASET_ACCESS_REQUIRED',
+        'Approved dataset access is required to run this query.',
+      );
+    }
+
+    if (!parsed.success) {
+      return NextResponse.json(
+        {
+          errors: Object.fromEntries(
+            parsed.error.issues.map((issue) => [issue.path.join('.') || 'request', issue.message]),
+          ),
+        },
+        { status: 400 },
+      );
+    }
+
+    const remaining = await remainingQueries(loaded, principal, access);
+    if (remaining === 0) {
+      return errorResponse(
+        429,
+        'QUERY_LIMIT_REACHED',
+        'You have used all queries included with your access to this dataset.',
+      );
+    }
+
+    const retrievedAt = new Date();
+    const database = await openImportedDatabase(
+      loaded.imported.id,
+      new Uint8Array(loaded.imported.sourceDatabase as Uint8Array),
+    );
+
+    let result: ImportedQueryResult;
+    try {
+      result = executeImportedQuery(database, catalogFor(loaded), parsed.data, {
+        datasetName: loaded.dataset.name,
+        provider: loaded.dataset.provider,
+        sourceFilename: loaded.imported.sourceFilename,
+        importedAt: loaded.imported.createdAt,
+        retrievedAt,
+      });
+    } catch (error) {
+      if (error instanceof ImportedQueryValidationError) {
+        return NextResponse.json({ errors: { [error.field]: error.message } }, { status: 400 });
+      }
+      throw error;
+    }
+
+    const storedQuery: ImportedStoredQuery = {
+      kind: 'imported_table',
+      table: parsed.data.table,
+      filters: parsed.data.filters,
+    };
+    if (loaded.dataset.ownerId) {
+      await recordUsageLedgerEntry({
+        userId: principal.userId,
+        providerId: loaded.dataset.ownerId,
+        datasetId: loaded.dataset.id,
+        datasetKey: loaded.dataset.stableKey,
+        datasetName: loaded.dataset.name,
+        provider: loaded.dataset.provider,
+        query: storedQuery as Prisma.InputJsonValue,
+        response:
+          result.status === 'matched'
+            ? {
+                kind: 'imported_table',
+                status: 'matched',
+                responseMode: result.responseMode,
+                table: result.table,
+                matchCount: result.matchCount,
+                recordCount:
+                  result.responseMode === 'raw_records'
+                    ? result.records.length
+                    : result.responseMode === 'answer_only'
+                      ? 1
+                      : 0,
+                truncated: result.responseMode === 'raw_records' ? result.truncated : false,
+                provenance: result.provenance,
+              }
+            : {
+                kind: 'imported_table',
+                status: result.status,
+                responseMode: result.responseMode,
+                table: result.table,
+                error: result.error,
+              },
+        outcome: result.status === 'matched' ? 'success' : 'failed',
+        queryPriceCents: loaded.dataset.queryPriceCents,
+        revenueSharePercent: loaded.dataset.revenueSharePercent,
+        retrievedAt,
+      });
+    }
+
+    // no_match / ambiguous are valid query outcomes, so they return 200 with a status field.
+    return NextResponse.json(ImportedQueryResult.parse(result));
+  } catch {
+    return internalError();
+  }
+}

+/**
+ * Loads the imported-dataset schema for `slug`. Renders nothing when the
+ * dataset has no imported source (e.g. the automotive demo dataset).
+ */
+export function ImportedDatasetQuery({ slug, reloadKey }: { slug: string; reloadKey: string }) {
+  const [schema, setSchema] = useState<ImportedDatasetSchemaType | null>(null);
+
+  // biome-ignore lint/correctness/useExhaustiveDependencies: reloadKey intentionally re-fetches access after sign-in or approval changes.
+  useEffect(() => {
+    let cancelled = false;
+    apiFetch(`/api/datasets/${encodeURIComponent(slug)}/query`, { schema: ImportedDatasetSchema })
+      .then((value) => {
+        if (!cancelled) setSchema(value);
+      })
+      .catch(() => {
+        if (!cancelled) setSchema(null);
+      });
+    return () => {
+      cancelled = true;
+    };
+  }, [slug, reloadKey]);
+
+  if (!schema) return null;
+
+  const canQuery =
+    schema.access === 'owner' || schema.access === 'admin' || schema.access === 'approved';
+
+  return (
+    <section aria-label="Query this dataset" className="space-y-4">
+      {canQuery ? (
+        <QueryForm slug={slug} schema={schema} />
+      ) : (
+        <AccessNotice signedIn={schema.access !== 'sign_in_required'} />
+      )}
+    </section>
+  );

+export class ImportedQueryValidationError extends Error {
+  constructor(
+    message: string,
+    readonly field: 'table' | 'filters',
+  ) {
+    super(message);
+    this.name = 'ImportedQueryValidationError';
+  }
+}

+export type ImportedTableCatalog = {
+  name: string;
+  columns: string[];
+  queryableColumns: string[];

+// Small in-process cache so repeated queries don't re-parse the file. Keyed by
+// import id + byte length so a re-import is never served stale.
+const DATABASE_CACHE_LIMIT = 8;
+const databaseCache = new Map<string, Database>();
+
+export async function openImportedDatabase(cacheKey: string, bytes: Uint8Array): Promise<Database> {
+  const key = `${cacheKey}:${bytes.byteLength}`;
+  const cached = databaseCache.get(key);
+  if (cached) {
+    databaseCache.delete(key);
+    databaseCache.set(key, cached);
+    return cached;
+  }
+  const SQL = await loadSqlite();
+  const database = new SQL.Database(bytes);
+  databaseCache.set(key, database);
+  while (databaseCache.size > DATABASE_CACHE_LIMIT) {
+    const oldest = databaseCache.keys().next().value;
+    if (oldest === undefined) break;
+    databaseCache.get(oldest)?.close();
+    databaseCache.delete(oldest);
+  }
+  return database;
+}
+
+function toCell(value: SqlValue): string | number | boolean | null {
+  if (value === null || typeof value === 'string' || typeof value === 'number') return value;
+  return null; // binary blobs are never returned
+}
+
+function selectAll(
+  database: Database,
+  sql: string,
+  params: SqlValue[],
+): { columns: string[]; values: SqlValue[][] } {
+  const [result] = database.exec(sql, params);
+  return result ? { columns: result.columns, values: result.values } : { columns: [], values: [] };
+}
+
+function toRecords(result: { columns: string[]; values: SqlValue[][] }): ImportedRecord[] {
+  return result.values.map((row) =>
+    Object.fromEntries(result.columns.map((column, index) => [column, toCell(row[index] ?? null)])),
+  );
+}
+
+function sqliteObjectColumns(database: Database, name: string): string[] | null {
+  const kind = selectAll(
+    database,
+    "SELECT type FROM sqlite_master WHERE name = ? AND type IN ('table', 'view')",
+    [name],
+  );
+  if (kind.values.length === 0) return null;
+  return selectAll(database, `PRAGMA table_info(${quoteIdentifier(name)})`, []).values.map((row) =>
+    String(row[1]),
+  );
+}
+
+function whereClause(filters: Record<string, ImportedFilterValue>) {
+  const clauses: string[] = [];
+  const params: SqlValue[] = [];
+  for (const [column, value] of Object.entries(filters)) {
+    if (typeof value === 'string') {
+      clauses.push(`TRIM(CAST(${quoteIdentifier(column)} AS TEXT)) = ? COLLATE NOCASE`);
+      params.push(value.trim());
+    } else {
+      clauses.push(`${quoteIdentifier(column)} = ?`);
+      params.push(typeof value === 'boolean' ? (value ? 1 : 0) : value);
+    }
+  }
+  return { sql: clauses.join(' AND '), params };
+}
+
+const SOURCE_FIELD_ALIASES: Record<keyof Omit<ImportedCitedSource, 'sourceId'>, string[]> = {
+  publisher: ['publisher', 'source_publisher', 'organization'],
+  title: ['title', 'source_title', 'name'],
+  url: ['url', 'source_url', 'link'],
+  accessedDate: ['accessed_date', 'accessed_at', 'retrieved_at'],
+};
+
+// Convention: if the queried table has a `source_id` column and the database
+// contains a `sources` table keyed by `source_id`, cite those sources.
+function citedSources(
+  database: Database,
+  table: string,
+  where: { sql: string; params: SqlValue[] },
+  tableColumns: string[],
+): ImportedCitedSource[] {
+  if (!tableColumns.includes('source_id')) return [];
+  const sourceColumns = sqliteObjectColumns(database, 'sources');
+  if (!sourceColumns?.includes('source_id')) return [];
+
+  const ids = selectAll(
+    database,
+    `SELECT DISTINCT source_id FROM ${quoteIdentifier(table)} WHERE ${where.sql} AND source_id IS NOT NULL LIMIT 20`,
+    where.params,
+  ).values.map((row) => String(row[0]));
+  if (ids.length === 0) return [];
+
+  const rows = toRecords(
+    selectAll(
+      database,
+      `SELECT * FROM sources WHERE source_id IN (${ids.map(() => '?').join(', ')})`,
+      ids,
+    ),
+  );
+  return rows.map((row) => {
+    const cited: ImportedCitedSource = { sourceId: String(row.source_id) };
+    for (const [field, aliases] of Object.entries(SOURCE_FIELD_ALIASES)) {
+      const alias = aliases.find((name) => typeof row[name] === 'string' && row[name] !== '');
+      if (alias) cited[field as keyof typeof SOURCE_FIELD_ALIASES] = String(row[alias]);
+    }
+    return cited;
+  });
+}

+export function executeImportedQuery(
+  database: Database,
+  catalog: ImportedTableCatalog[],
+  request: ImportedQueryRequest,
+  context: ImportedQueryContext,
+): ImportedQueryResult {
+  const table = catalog.find((entry) => entry.name === request.table);
+  if (!table || table.queryableColumns.length === 0) {
+    throw new ImportedQueryValidationError('This table is not available for queries.', 'table');
+  }
+  const tableColumns = sqliteObjectColumns(database, table.name);
+  if (!tableColumns) {
+    throw new ImportedQueryValidationError('This table is not available for queries.', 'table');
+  }
+  for (const column of Object.keys(request.filters)) {
+    if (!table.queryableColumns.includes(column) || !tableColumns.includes(column)) {
+      throw new ImportedQueryValidationError(
+        `"${column}" is not a permitted filter for ${table.name}.`,
+        'filters',
+      );
+    }
+  }
+
+  const identifier = quoteIdentifier(table.name);
+  const where = whereClause(request.filters);
+  const matchCount = Number(
+    selectAll(database, `SELECT COUNT(*) FROM ${identifier} WHERE ${where.sql}`, where.params)
+      .values[0]?.[0] ?? 0,
+  );
+  const base = { table: table.name, responseMode: request.responseMode, matchCount };
+
+  if (matchCount === 0) {
+    return {
+      ...base,
+      status: 'no_match',
+      error: { code: 'NO_MATCH', message: 'No records match these filters.' },
+    };
+  }
+  if (request.responseMode === 'answer_only' && matchCount > 1) {
+    return {
+      ...base,
+      status: 'ambiguous',
+      error: {
+        code: 'AMBIGUOUS_MATCH',
+        message: `${matchCount} records match. Add filters to narrow to one, or use summarized data.`,
+      },
+    };
+  }
+
+  const provenance = {
+    dataset: context.datasetName,
+    provider: context.provider,
+    sourceFilename: context.sourceFilename,
+    importedAt: context.importedAt.toISOString(),
+    retrievedAt: context.retrievedAt.toISOString(),
+    citedSources: citedSources(database, table.name, where, tableColumns),
+  };
+
+  if (request.responseMode === 'answer_only') {
+    const [answer] = toRecords(
+      selectAll(database, `SELECT * FROM ${identifier} WHERE ${where.sql} LIMIT 1`, where.params),
+    );
+    return {
+      ...base,
+      status: 'matched',
+      responseMode: 'answer_only',
+      answer: answer ?? {},
+      provenance,
+    };
+  }
+
+  if (request.responseMode === 'summarized_data') {
+    const summary = tableColumns.slice(0, IMPORTED_SUMMARY_COLUMN_LIMIT).map((column) => {
+      const col = quoteIdentifier(column);
+      const values = selectAll(
+        database,
+        `SELECT ${col}, COUNT(*) FROM ${identifier} WHERE ${where.sql} GROUP BY ${col} ORDER BY COUNT(*) DESC, ${col} LIMIT ${IMPORTED_SUMMARY_VALUE_LIMIT + 1}`,
+        where.params,
+      ).values;
+      return {
+        column,
+        distinctValues: values
+          .slice(0, IMPORTED_SUMMARY_VALUE_LIMIT)
+          .map(([value, count]) => ({ value: toCell(value ?? null), count: Number(count) })),
+        truncated: values.length > IMPORTED_SUMMARY_VALUE_LIMIT,
+      };
+    });
+    return { ...base, status: 'matched', responseMode: 'summarized_data', summary, provenance };
+  }
+
+  const records = toRecords(
+    selectAll(
+      database,
+      `SELECT * FROM ${identifier} WHERE ${where.sql} LIMIT ${IMPORTED_RAW_RECORD_LIMIT}`,
+      where.params,
+    ),
+  );
+  return {
+    ...base,
+    status: 'matched',
+    responseMode: 'raw_records',
+    records,
+    extractionLimit: IMPORTED_RAW_RECORD_LIMIT,
+    truncated: matchCount > records.length,
+    provenance,
+  };
+}

+  rowCount: number;

+export async function catalogSqliteFile(bytes: Uint8Array): Promise<CatalogEntry[]> {
+  const SQL = await loadSqlite();
+  const database = new SQL.Database(bytes);
+  try {
+    const names =
+      database.exec(
+        "SELECT name FROM sqlite_master WHERE type IN ('table', 'view') AND name NOT LIKE 'sqlite_%' ORDER BY name",
+      )[0]?.values ?? [];
+    return names.map(([name]) => {
+      const identifier = quoteIdentifier(String(name));
+      const columns = (database.exec(`PRAGMA table_info(${identifier})`)[0]?.values ?? []).map(
+        (row) => ({
+          name: String(row[1]),
+          type: String(row[2] ?? ''),
+          notNull: row[3] === 1,
+          primaryKey: typeof row[5] === 'number' && row[5] > 0,
+        }),
+      );
+      const rowCount = Number(
+        database.exec(`SELECT COUNT(*) FROM ${identifier}`)[0]?.values[0]?.[0] ?? 0,
+      );
+      return { name: String(name), rowCount, columns };
+    });
+  } finally {
+    database.close();
+  }
